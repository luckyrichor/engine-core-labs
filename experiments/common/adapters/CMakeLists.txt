# Only consume pinned local builds with successful receipts, never system packages.
function(ecl_prefix name row out)
  file(STRINGS "${CMAKE_SOURCE_DIR}/third_party/versions.md" pin_rows ENCODING UTF-8)
  set(pin "")
  foreach(line IN LISTS pin_rows)
    string(FIND "${line}" "| ${row} |" at)
    if(at EQUAL 0)
      set(pin "${line}")
      break()
    endif()
  endforeach()
  if(NOT pin)
    message(FATAL_ERROR "Missing baseline pin: ${row}")
  endif()
  string(REPLACE "|" ";" cells "${pin}")
  list(GET cells 3 tag)
  list(GET cells 4 commit)
  string(STRIP "${tag}" tag)
  string(STRIP "${commit}" commit)
  if(tag STREQUAL "无标签")
    set(tag "${commit}")
  endif()
  set(prefix "${ECL_DATA}/third_party/install/${name}-${tag}")
  if(ECL_DATA AND EXISTS "${prefix}/build-info.json")
    file(READ "${prefix}/build-info.json" receipt)
    string(JSON actual GET "${receipt}" commit)
    string(JSON actual_name GET "${receipt}" name)
    string(FIND "${actual}" "${commit}" match)
    if(NOT match EQUAL 0 OR NOT actual_name STREQUAL name)
      message(FATAL_ERROR "Invalid baseline receipt: ${prefix}")
    endif()
    set(${out} "${prefix}" PARENT_SCOPE)
  else()
    set(${out} "" PARENT_SCOPE)
    message(STATUS "Baseline ${name} unavailable; adapter test will be SKIPPED (run scripts/third_party/build.sh)")
  endif()
endfunction()
function(adapter_test name)
  add_test(NAME "${name}" COMMAND ${Python3_EXECUTABLE}
    "${CMAKE_CURRENT_SOURCE_DIR}/check.py" ${ARGN})
  set_tests_properties("${name}" PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
    ENVIRONMENT "PYTHONPATH=${CMAKE_SOURCE_DIR}" SKIP_RETURN_CODE 77 TIMEOUT 120)
endfunction()
function(adapter_skip name)
  adapter_test("adapter_${name}_unavailable" skip "${name}")
endfunction()

function(alloc_target suffix definition label)
  add_executable("alloc_replay_${suffix}" alloc_replay.cpp alloc_identity.cpp)
  target_compile_definitions("alloc_replay_${suffix}" PRIVATE "${definition}" "ECL_ALLOC_NAME=\"${label}\"")
  target_link_libraries("alloc_replay_${suffix}" PRIVATE ${CMAKE_DL_LIBS})
  adapter_test("adapter_alloc_${suffix}" alloc "$<TARGET_FILE:alloc_replay_${suffix}>")
endfunction()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  alloc_target(libc ECL_ALLOC_LIBC glibc)
else()
  alloc_target(libc ECL_ALLOC_LIBC system-malloc)
endif()
alloc_target(local ECL_ALLOC_LOCAL variable-heap)
ecl_prefix(jemalloc jemalloc jemalloc_prefix)
ecl_prefix(mimalloc mimalloc mimalloc_prefix)
ecl_prefix(gperftools "tcmalloc（gperftools）" gperf_prefix)
ecl_prefix(google-tcmalloc "tcmalloc（google/tcmalloc）" google_prefix)
foreach(name jemalloc mimalloc gperftools)
  if(name STREQUAL "gperftools")
    set(prefix "${gperf_prefix}")
    set(library tcmalloc)
  else()
    set(prefix "${${name}_prefix}")
    set(library "${name}")
  endif()
  if(prefix)
    if(name STREQUAL "mimalloc")
      file(GLOB mi_configs LIST_DIRECTORIES true "${prefix}/lib/cmake/mimalloc*")
      find_package(mimalloc REQUIRED CONFIG PATHS ${mi_configs} NO_DEFAULT_PATH)
      set(ecl_mimalloc_library mimalloc)
    else()
      find_library(ecl_${name}_library NAMES "${library}" PATHS "${prefix}/lib" NO_DEFAULT_PATH REQUIRED)
    endif()
    string(TOUPPER "${name}" definition)
    set(label "${name}")
    if(name STREQUAL "gperftools")
      set(label tcmalloc-gperftools)
    endif()
    alloc_target("${name}" "ECL_ALLOC_${definition}" "${label}")
    if(name STREQUAL "jemalloc")
      file(READ "${prefix}/build-info.json" receipt)
      string(JSON expected_version GET "${receipt}" tag)
      target_compile_definitions(alloc_replay_jemalloc PRIVATE "ECL_ALLOC_VERSION=\"${expected_version}\"")
    endif()
    target_include_directories("alloc_replay_${name}" SYSTEM PRIVATE "${prefix}/include")
    target_link_libraries("alloc_replay_${name}" PRIVATE "${ecl_${name}_library}")
    set_target_properties("alloc_replay_${name}" PROPERTIES BUILD_RPATH "${prefix}/lib")
  else()
    adapter_skip("${name}")
  endif()
endforeach()
if(google_prefix AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
  include("${google_prefix}/google-tcmalloc.cmake")
  alloc_target(google ECL_ALLOC_GOOGLE tcmalloc-google)
  target_link_libraries(alloc_replay_google PRIVATE ecl_google_tcmalloc)
  target_link_options(alloc_replay_google PRIVATE -Wl,--export-dynamic)
else()
  adapter_skip(google-tcmalloc)
endif()
