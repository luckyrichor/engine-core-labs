cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(engine_s8 LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  enable_testing()
endif()
find_package(Threads REQUIRED)
find_package(Python3 3.12 COMPONENTS Interpreter REQUIRED)
add_executable(phase4_bench benchmark.cpp)
target_link_libraries(phase4_bench PRIVATE Threads::Threads)
if(WIN32)
  target_compile_definitions(phase4_bench PRIVATE _WIN32_WINNT=0x0601 NOMINMAX)
else()
  target_compile_options(phase4_bench PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME phase4_layout_contract COMMAND ${Python3_EXECUTABLE}
  "${CMAKE_CURRENT_SOURCE_DIR}/check.py" "$<TARGET_FILE:phase4_bench>")
get_filename_component(ecl_root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set_tests_properties(phase4_layout_contract PROPERTIES WORKING_DIRECTORY "${ecl_root}"
  ENVIRONMENT "PYTHONPATH=${ecl_root}" TIMEOUT 120)
