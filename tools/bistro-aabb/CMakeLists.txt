            AND EXISTS "${ECL_DATA}/third_party/ufbx-v0.23.1/ufbx.h")
  enable_language(C)
  add_library(bistro_ufbx STATIC "${ECL_DATA}/third_party/ufbx-v0.23.1/ufbx.c")

  add_executable(bistro_aabb main.cpp)
  target_link_libraries(bistro_aabb PRIVATE bistro_ufbx)
  target_include_directories(bistro_aabb PRIVATE "${PROJECT_SOURCE_DIR}/experiments/common")
  set_target_properties(bistro_aabb PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${ECL_DATA}/tools/bin")
