cmake_minimum_required(VERSION 3.20)
project(engine_core_labs LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ENGINE_SANITIZER "none" CACHE STRING "none, address, or thread")
if(ENGINE_SANITIZER STREQUAL "address")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  add_link_options(-fsanitize=address,undefined -no-pie)
elseif(ENGINE_SANITIZER STREQUAL "thread")
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer -fno-pie)
  add_link_options(-fsanitize=thread -no-pie)
elseif(NOT ENGINE_SANITIZER STREQUAL "none")
  message(FATAL_ERROR "unknown sanitizer")
endif()
enable_testing()
add_subdirectory(experiments/l1-job-system)
add_subdirectory(experiments/l2-allocators)
add_subdirectory(experiments/l3-resource-loading)

find_package(Python3 3.12 COMPONENTS Interpreter REQUIRED)
add_test(NAME bench_unittest
  COMMAND ${Python3_EXECUTABLE} -m unittest discover -s bench/tests -v)
set_tests_properties(bench_unittest PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")

add_executable(workload_reader_test bench/tests/workload_reader.cpp)
target_include_directories(workload_reader_test PRIVATE "${CMAKE_SOURCE_DIR}/experiments/common")
add_test(NAME bench_workload_cpp
  COMMAND ${Python3_EXECUTABLE} -m bench.tests.test_workload_cpp $<TARGET_FILE:workload_reader_test>)
set_tests_properties(bench_workload_cpp PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
